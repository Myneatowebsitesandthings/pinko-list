document.addEventListener("DOMContentLoaded", function () {
  const searchInput = document.getElementById("businessSearch");
  const resultsBox = document.getElementById("businessResults");

  const nameHeader = document.getElementById("businessName");
  const descriptionParagraph = document.getElementById("businessDescription");

  const issuesList = document.getElementById("issuesList");
  const peopleList = document.getElementById("peopleList");
  const subsidiariesList = document.getElementById("subsidiariesList");

  if (!searchInput || !resultsBox) {
    console.error("Missing businessSearch or businessResults in HTML.");
    return;
  }

  let companiesData = [];
  let filteredResults = [];
  let activeIndex = 0;

  fetch("companies.json")
    .then(r => r.json())
    .then(data => {
      companiesData = (data || []).sort((a, b) =>
        (a.name || "").localeCompare(b.name || "")
      );
      filteredResults = companiesData;
      renderResults(filteredResults);
    })
    .catch(err => console.error("Error loading companies.json:", err));

  function getRelatedNames(company) {
    let related = [];

    if (Array.isArray(company.subsidiaries)) {
      related = related.concat(company.subsidiaries);
    }
    if (Array.isArray(company.brands)) {
      related = related.concat(company.brands);
    }
    if (Array.isArray(company.other_companies)) {
      related = related.concat(company.other_companies);
    }

    return related;
  }

  function cleanText(text) {
    return (text || "").replace(/\[\d+\]/g, "");
  }

  function displayCompany(company, highlightTerm = "") {
    if (!company) return;

    nameHeader.textContent = company.name || "";
    descriptionParagraph.textContent = cleanText(company.description) || "No description available.";

    issuesList.innerHTML = "<p>No issues listed yet.</p>";
    peopleList.innerHTML = "<p>No associated people yet.</p>";
    subsidiariesList.innerHTML = "";

    const related = getRelatedNames(company);

    if (related.length === 0) {
      subsidiariesList.innerHTML = "<p>No subsidiaries listed.</p>";
      return;
    }

    related.forEach(item => {
      const p = document.createElement("p");
      p.textContent = item;

      if (highlightTerm && item.toLowerCase().includes(highlightTerm.toLowerCase())) {
        p.classList.add("highlight-brand");
      }

      subsidiariesList.appendChild(p);
    });
  }

  function filterCompanies(term) {
    const q = term.trim().toLowerCase();

    if (!q) return companiesData;

    return companiesData.filter(company => {
      const companyName = (company.name || "").toLowerCase();
      const companyMatch =
        companyName.startsWith(q) || companyName.includes(" " + q);

      const related = getRelatedNames(company);
      const relatedMatch = related.some(item =>
        (item || "").toLowerCase().startsWith(q) ||
        (item || "").toLowerCase().includes(" " + q)
      );

      return companyMatch || relatedMatch;
    });
  }

  function renderResults(results) {
    resultsBox.innerHTML = "";
    filteredResults = results;
    activeIndex = 0;

    if (!results.length) {
      resultsBox.style.display = "none";
      return;
    }

    results.slice(0, 12).forEach((company, index) => {
      const item = document.createElement("div");
      item.className = "business-result-item";
      if (index === 0) item.classList.add("active-result");

      item.textContent = company.name;

      item.addEventListener("mousedown", function (e) {
        e.preventDefault();
        selectCompany(company);
      });

      resultsBox.appendChild(item);
    });

    resultsBox.style.display = "block";
  }

  function updateActiveResult() {
    const items = resultsBox.querySelectorAll(".business-result-item");
    items.forEach((item, index) => {
      item.classList.toggle("active-result", index === activeIndex);
    });
  }

  function selectCompany(company) {
    searchInput.value = company.name;
    resultsBox.style.display = "none";

    const term = searchInput.value.trim();
    displayCompany(company, term);
  }

  searchInput.addEventListener("focus", function () {
    renderResults(filterCompanies(searchInput.value));
  });

  searchInput.addEventListener("input", function () {
    const results = filterCompanies(searchInput.value);
    renderResults(results);
  });

  searchInput.addEventListener("keydown", function (e) {
    const visibleResults = filteredResults.slice(0, 12);

    if (!visibleResults.length) return;

    if (e.key === "ArrowDown") {
      e.preventDefault();
      activeIndex = Math.min(activeIndex + 1, visibleResults.length - 1);
      updateActiveResult();
      return;
    }

    if (e.key === "ArrowUp") {
      e.preventDefault();
      activeIndex = Math.max(activeIndex - 1, 0);
      updateActiveResult();
      return;
    }

    if (e.key === "Enter") {
      e.preventDefault();
      selectCompany(visibleResults[activeIndex]);
    }
  });

  document.addEventListener("click", function (e) {
    if (!e.target.closest(".business-search-wrap")) {
      resultsBox.style.display = "none";
    }
  });
});
